{{- /* Splitst content op
in secties en kiest per sectie een brick via params.sections (has/not_has/is/not_is), of geforceerd via ---.naam; ---.naam op een lege sectie is een include uit content//bricks/. Roep aan via partialCached. */ -}} {{- $page := . -}} {{- $rules := site.Params.sections | default slice -}} {{- /* ---.small wordt door goldmark een paragraaf i.p.v. een
; vouw 'm terug tot een divider met brick-naam. */ -}} {{- $content := partial "inline_attributes.html" $page.Content -}} {{- $content = replaceRE `

(?:—|—|-{3,})\s*\.([a-zA-Z0-9][a-zA-Z0-9_-]*)

` "
" $content -}} {{- $sections := slice -}} {{- range $part := split $content "
" -}} {{- with trim $part "\n\r\t " -}} {{- $sections = $sections | append . -}} {{- end -}} {{- end -}} {{- $count := len $sections -}} {{- $out := slice -}} {{- range $i, $html := $sections -}} {{- $name := "wide" -}} {{- $matched := false -}} {{- $forced := false -}} {{- if hasPrefix $html ".*` "$1" $html -}} {{- $html = trim (strings.TrimPrefix (printf "" $name) $html) "\n\r\t " -}} {{- $matched = true -}} {{- $forced = true -}} {{- /* Genoemd maar leeg: een include, die zijn eigen hoisting-pas krijgt. */ -}} {{- if not $html -}} {{- with site.GetPage (printf "/bricks/%s" $name) -}} {{- $html = trim (partial "inline_attributes.html" .Content) "\n\r\t " -}} {{- else -}} {{- warnf "section_map: `---.%s` opens an empty section and there is no content//bricks/%s.md to fill it" $name $name -}} {{- end -}} {{- end -}} {{- end -}} {{- range $rule := $rules -}} {{- if not $matched -}} {{- $match := true -}} {{- $tested := false -}} {{- with $rule.has -}} {{- $tested = true -}} {{- $match = and $match (partial "section_has.html" (dict "Html" $html "Selectors" .)) -}} {{- end -}} {{- with $rule.not_has -}} {{- $tested = true -}} {{- $match = and $match (not (partial "section_has.html" (dict "Html" $html "Selectors" .))) -}} {{- end -}} {{- with $rule.is -}} {{- $tested = true -}} {{- $match = and $match (partial "section_position.html" (dict "Selector" . "Index" $i "Count" $count)) -}} {{- end -}} {{- with $rule.not_is -}} {{- $tested = true -}} {{- $match = and $match (not (partial "section_position.html" (dict "Selector" . "Index" $i "Count" $count))) -}} {{- end -}} {{- if not $tested -}} {{- warnf "section_map: rule %q sets no condition and would claim every section — skipped" $rule.title -}} {{- else if and $match $rule.title -}} {{- $name = $rule.title -}} {{- $matched = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not (templates.Exists (printf "_partials/sections/%s.html" $name)) -}} {{- warnf "section_map: missing partial %q, falling back to sections/wide" $name -}} {{- $name = "wide" -}} {{- end -}} {{- $out = $out | append (dict "Name" $name "Content" $html "Index" $i "Forced" $forced) -}} {{- end -}} {{- /* Image-bricks zigzaggen tussen getint en wit band; startfase hangt af van de openingssectie (na intro getint, na header wit). */ -}} {{- $pair := slice "image" "image-alt" -}} {{- $tinted := false -}} {{- with $out -}} {{- $tinted = eq (index . 0).Name "intro" -}} {{- end -}} {{- $zigzag := slice -}} {{- range $out -}} {{- $section := . -}} {{- if in $pair $section.Name -}} {{- /* Een geforceerde face verankert de fase opnieuw voor wat erna komt. */ -}} {{- if $section.Forced -}} {{- $tinted = eq $section.Name "image-alt" -}} {{- end -}} {{- $section = merge $section (dict "Name" (cond $tinted "image-alt" "image")) -}} {{- $tinted = not $tinted -}} {{- end -}} {{- $zigzag = $zigzag | append $section -}} {{- end -}} {{- return $zigzag -}}

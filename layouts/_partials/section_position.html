{{- /* Positie van een sectie op de pagina: :first/:last/:only-child, :nth-child(n), evt. in :not(…); onbekende selector geeft true (niet blokkeren op een typo). */ -}} {{- $selector := trim .Selector " " -}} {{- $i := .Index -}} {{- $n := .Count -}} {{- $negate := false -}} {{- if hasPrefix $selector ":not(" -}} {{- $negate = true -}} {{- $selector = trim (strings.TrimSuffix ")" (strings.TrimPrefix ":not(" $selector)) " " -}} {{- end -}} {{- $at := false -}} {{- $known := true -}} {{- if eq $selector ":first-child" -}} {{- $at = eq $i 0 -}} {{- else if eq $selector ":last-child" -}} {{- $at = eq $i (sub $n 1) -}} {{- else if eq $selector ":only-child" -}} {{- $at = eq $n 1 -}} {{- else if findRE `^:nth-child\(\s*\d+\s*\)$` $selector 1 -}} {{- $at = eq $i (sub (int (index (findRE `\d+` $selector 1) 0)) 1) -}} {{- else -}} {{- $known = false -}} {{- warnf "section_map: unsupported position %q in params.sections" .Selector -}} {{- end -}} {{- $result := true -}} {{- if $known -}}{{- $result = ne $at $negate -}}{{- end -}} {{- return $result -}}

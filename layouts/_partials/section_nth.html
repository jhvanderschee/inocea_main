{{- /* Sectie-html vanaf de nde top-level child; losse media telt niet mee als child (voor section_has :nth-child(n)). */ -}} {{- $rest := trim (string .Html) " \n\r\t" -}} {{- $n := int .N -}} {{- $void := `(?is)^<(?:img|hr|br|input|source|col|area|embed|track|wbr)\b[^>]*>\s*` -}} {{- $media := `(?is)^(?:

\s*(?:]*>\s*)?]*>\s*(?:\s*)?

|)\s*` -}} {{- $skip := `(?s)^(?:]*>\s*)+` -}} {{- if gt $n 1 -}} {{- range seq (sub $n 1) -}} {{- $rest = replaceRE $media "" (replaceRE $skip "" $rest) -}} {{- if hasPrefix $rest "<" -}} {{- if findRE $void $rest 1 -}} {{- $rest = replaceRE $void "" $rest -}} {{- else if findRE `(?is)^<[a-zA-Z][\w-]*[\s/>]` $rest 1 -}} {{- $name := replaceRE `(?is)^<([a-zA-Z][\w-]*)[\s/>].*$` "$1" $rest -}} {{- $rest = replaceRE (printf `(?is)^.*?\s*` $name) "" $rest -}} {{- else -}} {{- $rest = "" -}} {{- end -}} {{- else -}} {{- /* losse tekst op het top-level telt ook als één child */ -}} {{- $rest = replaceRE `(?s)^[^<]*` "" $rest -}} {{- end -}} {{- $rest = trim $rest " \n\r\t" -}} {{- end -}} {{- end -}} {{- /* Nog één keer, ook voor N=1 dat de loop overslaat. */ -}} {{- $rest = trim (replaceRE $media "" (replaceRE $skip "" $rest)) " \n\r\t" -}} {{- return $rest -}}

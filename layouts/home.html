{{ define "content" }} {{- /* Secties komen uit de lopende content: elke `---` op een eigen regel is een grens. Elke section krijgt: 1. Eén class uit `site.Params.rules`. De regels worden van boven naar beneden afgelopen en de eerste die volledig klopt wint. Een regel kan `has`, `not_has`, `is` en `not_is` zetten; alles wat gezet is moet gelden. Een regel zonder enige conditie zou elke sectie claimen en wordt met een waarschuwing overgeslagen. `has` / `not_has` gaan door `section_has.html`: een komma-gescheiden lijst van een tag, een `.class` of een `#id`, elk optioneel met `:first-child`, `:last-child` of `:nth-child(n)`. Alles in de lijst moet aanwezig zijn, dus `not_has` is de exacte ontkenning van `has`. `is` / `not_is` gaan door `section_position.html`: de plek van de sectie op de pagina — `:first-child`, `:last-child`, `:only-child`, `:nth-child(n)`, eventueel in `:not(…)`. 2. has- voor elke shortcode die in de markdown van die sectie staat. */ -}} {{- $rules := site.Params.rules | default slice -}} {{- $chunks := split (replaceRE `(?m)^-{3,}[ \t]*$` "" .RawContent) "" -}} {{- $sections := slice -}} {{- range $chunks -}} {{- with trim . " \n\t\r" -}}{{- $sections = $sections | append . -}}{{- end -}} {{- end -}} {{- $count := len $sections -}} {{- range $i, $md := $sections -}} {{- $html := partial "content" ($md | $.RenderString (dict "display" "block")) -}} {{- $s := printf "%s" $html -}} {{- $classes := slice -}} {{- range $rule := $rules -}} {{- $match := true -}} {{- $tested := false -}} {{- with $rule.has -}} {{- $tested = true -}} {{- $match = and $match (partial "section_has.html" (dict "Html" $s "Selectors" .)) -}} {{- end -}} {{- with $rule.not_has -}} {{- $tested = true -}} {{- $match = and $match (not (partial "section_has.html" (dict "Html" $s "Selectors" .))) -}} {{- end -}} {{- with $rule.is -}} {{- $tested = true -}} {{- $match = and $match (partial "section_position.html" (dict "Selector" . "Index" $i "Count" $count)) -}} {{- end -}} {{- with $rule.not_is -}} {{- $tested = true -}} {{- $match = and $match (not (partial "section_position.html" (dict "Selector" . "Index" $i "Count" $count))) -}} {{- end -}} {{- if not $tested -}} {{- warnf "rules: regel %q stelt geen conditie en zou elke sectie claimen — overgeslagen" $rule.title -}} {{- else if and $match $rule.title -}} {{- $classes = $classes | append $rule.title -}} {{- break -}} {{- end -}} {{- end -}} {{- range findRE `\{\{[<%]-?\s*/?\s*[a-zA-Z0-9_-]+` $md -}} {{- $c := printf "has-%s" (replaceRE `^\{\{[<%]-?\s*/?\s*` "" .) -}} {{- if not (in $classes $c) -}}{{- $classes = $classes | append $c -}}{{- end -}} {{- end -}}
{{- $html -}}
{{- end -}} {{ end }}

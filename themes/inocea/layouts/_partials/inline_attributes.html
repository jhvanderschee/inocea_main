{{- /* Kramdown-achtige inline attribute lists: [tekst](url){:.class #id} of ![](img){:.class} wordt class/id op de tag; moet direct aansluiten, geen ruimte ertussen. */ -}} {{- $html := string . -}} {{- /* Getemperd i.p.v. lazy: Go-regexp kent geen lookahead, en .*? zou over een eerdere heen backtracken naar een latere {:. */ -}} {{- $anchor := `]*>(?:[^<]|<[^/]|])*` -}} {{- range $match := findRE (printf `(?:%s|]*>)\{:[^}]*\}` $anchor) $html -}} {{- $ial := index (findRE `\{:[^}]*\}$` $match 1) 0 -}} {{- $el := strings.TrimRight " \n\r\t" (strings.TrimSuffix $ial $match) -}} {{- $open := index (findRE `^<[a-zA-Z]+\b[^>]*>` $el 1) 0 -}} {{- $classes := slice -}} {{- $id := "" -}} {{- range $token := split (trim (strings.TrimSuffix "}" (strings.TrimPrefix "{:" $ial)) " \n\r\t") " " -}} {{- with trim $token " \n\r\t" -}} {{- if hasPrefix . "." -}} {{- range $name := split . "." -}} {{- with $name -}}{{- $classes = $classes | append . -}}{{- end -}} {{- end -}} {{- else if hasPrefix . "#" -}} {{- $id = strings.TrimPrefix "#" . -}} {{- else -}} {{- warnf "inline_attributes: ignoring %q — an attribute list takes `.class` and `#id`" . -}} {{- end -}} {{- end -}} {{- end -}} {{- $attrs := "" -}} {{- with $id -}}{{- $attrs = printf `%s id="%s"` $attrs . -}}{{- end -}} {{- with $classes -}}{{- $attrs = printf `%s class="%s"` $attrs (delimit . " ") -}}{{- end -}} {{- if and $attrs (findRE `\sclass="` $open 1) -}} {{- warnf "inline_attributes: %s already carries a class, so %q was dropped" $open $ial -}} {{- $attrs = "" -}} {{- end -}} {{- $replacement := $el -}} {{- with $attrs -}} {{- $replacement = replace $el $open (printf "%s%s>" (strings.TrimSuffix ">" $open) .) 1 -}} {{- end -}} {{- $html = replace $html $match $replacement 1 -}} {{- end -}} {{- return $html -}}
